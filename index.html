// Salvataggio sicuro delle foto su localStorage function savePhotos() { try { localStorage.setItem('plp_' + user.id, JSON.stringify(photos)); } catch (e) { if (e.name === 'QuotaExceededError') { toast('Spazio esaurito! Rimuovi alcune foto.'); } else { console.error(e); } } } // Funzione di rendering delle foto senza modificare l'array durante il ciclo function renderPhotos() { const container = document.getElementById('photoContainer'); container.innerHTML = ''; // Pulisce prima di renderizzare photos.forEach(photo => { const img = document.createElement('img'); img.src = photo.url; img.alt = photo.name || 'Foto'; container.appendChild(img); }); } // Inizializzazione dell'interfaccia utente function initUI() { // Controlla se l'UI è già stata inizializzata per evitare loop if (initUI.initialized) return; initUI.initialized = true; // Render delle foto renderPhotos(); // Event listener sicuri const addBtn = document.getElementById('addPhotoButton'); if (addBtn) { addBtn.addEventListener('click', () => { // Esempio: aggiunta di nuova foto const newPhoto = { url: 'nuova_foto.jpg', name: 'Nuova Foto' }; photos.push(newPhoto); renderPhotos(); savePhotos(); }); } const removeBtn = document.getElementById('removePhotoButton'); if (removeBtn) { removeBtn.addEventListener('click', () => { photos.pop(); // rimuove l'ultima foto renderPhotos(); savePhotos(); }); } } // Avvia UI initUI();
